🔗 Google Sheets como banco de dados
Com uma URL configurada, o Google Sheets vira a fonte de dados compartilhada do mapa: cada ponto criado, editado, movido ou apagado é enviado sozinho para a planilha (sem precisar clicar em nada), e o mapa busca sozinho o que outras pessoas cadastraram a cada 1 minuto. Todo mundo que abrir este arquivo com a mesma URL configurada enxerga os mesmos pontos.
Uso manual, só pra forçar agora — normalmente não precisa clicar nisso.
📋 Como configurar (uma vez só, por planilha)
1. Acesse sheets.google.com e crie uma planilha (o nome não importa, ex: GVC Mapa)
2. No menu, vá em Extensões → Apps Script
3. Apague o código existente e cole o código abaixo
4. Clique em Implantar → Nova implantação → App da Web
5. Em "Quem tem acesso" selecione Qualquer pessoa → Implantar
6. Copie a URL gerada e cole no campo acima → Salvar e sincronizar
7. Mande esta mesma URL pra quem mais for usar o mapa: eles colam no campo acima, no arquivo deles, e passam a ver os mesmos pontos
CÓDIGO DO APPS SCRIPT 
// Cole este código em Extensões → Apps Script da planilha que vai servir de banco de dados.
function getSpreadsheet_() { return SpreadsheetApp.getActiveSpreadsheet(); }

// ── Abas da planilha ──
//  Pontos            → um cadastro por linha
//  ResiduosCadastro  → um resíduo de um cadastro por linha, com preço por kg e se é receita ou custo
//  Residuos          → lista de tipos de resíduo (descrição)
//  Equipamentos      → lista de tipos de equipamento
const SHEET_NAME = 'Pontos';
const RC_SHEET_NAME = 'ResiduosCadastro';
const HEADERS = ['id','tipos','nome','razaoSocial','cat','cnpj','contatoNome','telefones','emails','logradouro','numero','bairro','cidade','uf','cep','obs','equipamentos','lat','lng','gmaps'];
const RC_HEADERS = ['id_cadastro','cadastro','residuo','preco_kg','natureza'];
const LISTAS = {
  residuos:     { aba: 'Residuos',     cabecalho: 'residuo' },
  equipamentos: { aba: 'Equipamentos', cabecalho: 'equipamento' },
};
const COLUNAS_NUMERICAS = { lat: true, lng: true, preco_kg: true };

// ── utilidades ──
function getAba_(nome, cabecalho) {
  const ss = getSpreadsheet_();
  let sheet = ss.getSheetByName(nome);
  if (!sheet) sheet = ss.insertSheet(nome);
  if (sheet.getLastRow() === 0) sheet.appendRow(cabecalho);
  return sheet;
}

// Grava várias linhas a partir da linha `inicio`, com colunas de texto em formato texto
// (evita que telefone, CEP ou algo começando com "=" vire número ou fórmula).
function escreverLinhas_(sheet, inicio, headers, linhas) {
  if (!linhas.length) return;
  headers.forEach((h, j) => {
    sheet.getRange(inicio, j + 1, linhas.length, 1).setNumberFormat(COLUNAS_NUMERICAS[h] ? '0.######' : '@');
  });
  sheet.getRange(inicio, 1, linhas.length, headers.length).setValues(linhas);
}

function reescreverAba_(sheet, headers, linhas) {
  sheet.clearContents();
  sheet.getRange(1, 1, 1, headers.length).setValues([headers]);
  escreverLinhas_(sheet, 2, headers, linhas);
}

function lerObjetos_(sheet) {
  const data = sheet.getDataRange().getValues();
  if (data.length <= 1) return [];
  const headers = data[0].map(String);
  return data.slice(1)
    .filter(r => String(r[0]) !== '')
    .map(r => { const o = {}; headers.forEach((h, i) => { if (h) o[h] = r[i]; }); return o; });
}

function juntar_(v, sep) { return Array.isArray(v) ? v.map(String).filter(x => x.trim() !== '').join(sep) : (v == null ? '' : String(v)); }
function separar_(v, re) { return String(v == null ? '' : v).split(re).map(x => x.trim()).filter(x => x !== ''); }
function json_(v, padrao) { if (typeof v !== 'string') return v || padrao; try { return v ? JSON.parse(v) : padrao; } catch (e) { return padrao; } }

function naturezaPadrao_(tipos) {
  const r = tipos.indexOf('reciclador') !== -1, d = tipos.indexOf('destinador') !== -1;
  return r && !d ? 'receita' : (d && !r ? 'custo' : '');
}

// ── aba Pontos (com migração automática do formato antigo) ──
function getSheet_() {
  const sheet = getAba_(SHEET_NAME, HEADERS);
  const atual = sheet.getRange(1, 1, 1, Math.max(sheet.getLastColumn(), 1)).getValues()[0].map(String).filter(h => h !== '');
  if (atual.join('|') !== HEADERS.join('|')) migrarPontos_(sheet);
  return sheet;
}

// Formato antigo: tipo (1 só), tel, email e resíduos/preços em JSON dentro da aba Pontos.
// Converte para o formato novo e move resíduos e preços para a aba ResiduosCadastro.
function migrarPontos_(sheet) {
  const antigos = lerObjetos_(sheet);
  const rc = getAba_(RC_SHEET_NAME, RC_HEADERS);
  const rcAtual = lerObjetos_(rc);
  const comRc = {};
  rcAtual.forEach(o => { comRc[String(o.id_cadastro)] = true; });
  const novasRc = [];
  const linhas = antigos.map(o => {
    const tipos = o.tipos ? separar_(o.tipos, /\s*,\s*/) : (o.tipo ? [String(o.tipo)] : []);
    const p = {
      id: o.id, tipos: tipos, nome: o.nome, razaoSocial: o.razaoSocial || '', cat: o.cat, cnpj: o.cnpj,
      contatoNome: o.contatoNome,
      telefones: o.telefones != null && o.telefones !== '' ? o.telefones : (o.tel || ''),
      emails: o.emails != null && o.emails !== '' ? o.emails : (o.email || ''),
      logradouro: o.logradouro, numero: o.numero, bairro: o.bairro, cidade: o.cidade, uf: o.uf, cep: o.cep,
      obs: o.obs, equipamentos: o.equipamentos || '', lat: o.lat, lng: o.lng, gmaps: o.gmaps,
    };
    if (!comRc[String(o.id)]) {
      const residuos = json_(o.residuos, []);
      const precos = json_(o.precos, {});
      const nat = naturezaPadrao_(tipos);
      if (Array.isArray(residuos)) residuos.forEach(r => {
        const preco = precos && precos[r] != null && precos[r] !== '' ? Number(precos[r]) : '';
        novasRc.push([String(o.id), String(o.nome || ''), String(r), preco, nat]);
      });
    }
    return pointToRow_(p);
  });
  reescreverAba_(sheet, HEADERS, linhas);
  if (novasRc.length) escreverLinhas_(rc, rc.getLastRow() + 1, RC_HEADERS, novasRc);
}

function findRow_(sheet, id) {
  const last = sheet.getLastRow();
  if (last < 2) return -1;
  const ids = sheet.getRange(2, 1, last - 1, 1).getValues();
  for (let i = 0; i < ids.length; i++) {
    if (String(ids[i][0]) === String(id)) return i + 2;
  }
  return -1;
}

function pointToRow_(p) {
  return HEADERS.map(h => {
    if (h === 'tipos') return juntar_(p.tipos && p.tipos.length ? p.tipos : (p.tipo ? [p.tipo] : []), ', ');
    if (h === 'telefones' || h === 'emails' || h === 'equipamentos') return juntar_(p[h], '; ');
    const v = p[h];
    return v !== undefined && v !== null ? v : '';
  });
}

// ── aba ResiduosCadastro ──
function linhasResiduos_(p) {
  const precos = p.precos || {}, naturezas = p.naturezas || {};
  return (p.residuos || []).map(r => [
    String(p.id), String(p.nome || ''), String(r),
    precos[r] != null && precos[r] !== '' ? Number(precos[r]) : '',
    naturezas[r] || '',
  ]);
}

function gravarResiduosDoCadastro_(p) {
  const rc = getAba_(RC_SHEET_NAME, RC_HEADERS);
  const outras = lerObjetos_(rc)
    .filter(o => String(o.id_cadastro) !== String(p.id))
    .map(o => RC_HEADERS.map(h => o[h] !== undefined ? o[h] : ''));
  reescreverAba_(rc, RC_HEADERS, outras.concat(linhasResiduos_(p)));
}

function apagarResiduosDoCadastro_(id) {
  const rc = getAba_(RC_SHEET_NAME, RC_HEADERS);
  const todas = lerObjetos_(rc);
  const outras = todas.filter(o => String(o.id_cadastro) !== String(id));
  if (outras.length !== todas.length) reescreverAba_(rc, RC_HEADERS, outras.map(o => RC_HEADERS.map(h => o[h] !== undefined ? o[h] : '')));
}

// ── listas (tipos de resíduo / tipos de equipamento) ──
function getLista_(nome) {
  const cfg = LISTAS[nome];
  const sheet = getAba_(cfg.aba, [cfg.cabecalho]);
  const last = sheet.getLastRow();
  if (last < 2) return [];
  return sheet.getRange(2, 1, last - 1, 1).getValues().map(r => String(r[0]).trim()).filter(x => x !== '');
}

function setLista_(nome, list) {
  const cfg = LISTAS[nome];
  const sheet = getAba_(cfg.aba, [cfg.cabecalho]);
  reescreverAba_(sheet, [cfg.cabecalho], list.map(x => [String(x)]));
}

// ── leitura ──
function doGet(e) {
  const lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    const action = e && e.parameter ? e.parameter.action : '';
    if (action === 'residuos') return jsonResponse_(getLista_('residuos'));
    if (action === 'lista' && LISTAS[e.parameter.nome]) return jsonResponse_(getLista_(e.parameter.nome));

    const sheet = getSheet_();
    const porCadastro = {};
    lerObjetos_(getAba_(RC_SHEET_NAME, RC_HEADERS)).forEach(o => {
      const id = String(o.id_cadastro);
      const x = porCadastro[id] || (porCadastro[id] = { residuos: [], precos: {}, naturezas: {} });
      const r = String(o.residuo);
      x.residuos.push(r);
      if (o.preco_kg !== '' && o.preco_kg != null) x.precos[r] = Number(o.preco_kg);
      if (o.natureza) x.naturezas[r] = String(o.natureza);
    });
    const rows = lerObjetos_(sheet).map(o => {
      o.id = String(o.id);
      o.tipos = separar_(o.tipos, /\s*,\s*/);
      o.tipo = o.tipos[0] || '';
      o.telefones = separar_(o.telefones, /\s*;\s*/);
      o.emails = separar_(o.emails, /\s*;\s*/);
      o.equipamentos = separar_(o.equipamentos, /\s*;\s*/);
      const x = porCadastro[o.id] || { residuos: [], precos: {}, naturezas: {} };
      o.residuos = x.residuos; o.precos = x.precos; o.naturezas = x.naturezas;
      return o;
    });
    return jsonResponse_(rows);
  } finally {
    lock.releaseLock();
  }
}

// ── escrita ──
// Ações aceitas no corpo do POST:
//  {action:'upsert', point:{...}}               → cria ou atualiza 1 cadastro (e os resíduos dele)
//  {action:'delete', id:'...'}                  → apaga 1 cadastro (e os resíduos dele)
//  {action:'listaAdd', nome:'residuos'|'equipamentos', valor:'...'}
//  {action:'listaDel', nome:'residuos'|'equipamentos', valor:'...'}
//  {action:'listaSet', nome:'residuos'|'equipamentos', list:[...]}  → só preenche se a lista estiver vazia
//  (aliases antigos: addResiduo / deleteResiduo / setResiduos)
//  [ {...}, {...} ]                             → substitui todos os cadastros (backup manual)
function doPost(e) {
  const lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    const body = JSON.parse(e.postData.contents);
    const a = body && body.action;

    let nomeLista = null, valor = null, op = null;
    if (a === 'listaAdd' || a === 'listaDel' || a === 'listaSet') { nomeLista = body.nome; valor = body.valor; op = a; }
    if (a === 'addResiduo')    { nomeLista = 'residuos'; valor = body.nome; op = 'listaAdd'; }
    if (a === 'deleteResiduo') { nomeLista = 'residuos'; valor = body.nome; op = 'listaDel'; }
    if (a === 'setResiduos')   { nomeLista = 'residuos'; op = 'listaSet'; }
    if (op) {
      if (!LISTAS[nomeLista]) return jsonResponse_({ ok: false, error: 'lista desconhecida' });
      let list = getLista_(nomeLista);
      if (op === 'listaAdd' && valor) {
        const v = String(valor).trim();
        if (!list.some(x => x.toLowerCase() === v.toLowerCase())) { list.push(v); setLista_(nomeLista, list); }
      } else if (op === 'listaDel' && valor) {
        list = list.filter(x => x !== String(valor));
        setLista_(nomeLista, list);
      } else if (op === 'listaSet' && Array.isArray(body.list)) {
        if (list.length === 0) { setLista_(nomeLista, body.list.map(String).filter(x => x.trim() !== '')); list = getLista_(nomeLista); }
      }
      return jsonResponse_({ ok: true, list: list });
    }

    const sheet = getSheet_();

    if (a === 'upsert' && body.point) {
      const p = body.point;
      const row = pointToRow_(p);
      const rowIdx = findRow_(sheet, p.id);
      escreverLinhas_(sheet, rowIdx === -1 ? sheet.getLastRow() + 1 : rowIdx, HEADERS, [row]);
      gravarResiduosDoCadastro_(p);
      return jsonResponse_({ ok: true });
    }

    if (a === 'delete' && body.id) {
      const rowIdx = findRow_(sheet, body.id);
      if (rowIdx !== -1) sheet.deleteRow(rowIdx);
      apagarResiduosDoCadastro_(body.id);
      return jsonResponse_({ ok: true });
    }

    if (Array.isArray(body)) {
      reescreverAba_(sheet, HEADERS, body.map(pointToRow_));
      const rc = getAba_(RC_SHEET_NAME, RC_HEADERS);
      reescreverAba_(rc, RC_HEADERS, [].concat.apply([], body.map(linhasResiduos_)));
      return jsonResponse_({ ok: true, count: body.length });
    }

    return jsonResponse_({ ok: false, error: 'ação desconhecida' });
  } finally {
    lock.releaseLock();
  }
}

function jsonResponse_(data) {
  return ContentService
    .createTextOutput(JSON.stringify(data))
    .setMimeType(ContentService.MimeType.JSON);
}
